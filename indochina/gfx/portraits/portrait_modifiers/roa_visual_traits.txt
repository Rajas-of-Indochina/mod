		dna_modifiers = {
			morph = {
				mode = add
				gene = gene_ainu_face
				template = ainu_face
				value = 1.0
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 50

						is_female = yes
						age >= 10
					}
				}
				trigger_else = { has_character_flag = ainu_tattoo_flag }
			}
		}
	}
