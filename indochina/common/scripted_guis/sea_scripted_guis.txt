	scope = character
	saved_scopes = {
		faith
	}
	effect = {
		remove_variable = selected_culture

armenian_struggle_realm_alignment_map = {
	effect = {
		set_global_variable = { name = custom_map_mode_changing value = 1 }
		set_global_variable = { name = custom_map_mode value = flag:armenian_struggle_realm_alignment_map }
		every_county = {
			every_county_province = { barony = { set_color_from_title = title:e_germany } }
			if = {
				limit = {
					OR = {
						kingdom = title:k_persia
						kingdom = title:k_thessalonika
						NOT = { any_county_struggle = { this = struggle:armenian_struggle } }
					}
				}
				every_county_province = { barony = { set_color_from_title = title:d_map_base_color } }
			}
			else_if = {
				limit = { holder = { has_trait = pro_roman } }
				every_county_province = { barony = { set_color_from_title = title:c_byzantion } }
			}
			else_if = {
				limit = { holder = { has_trait = pro_armenian } }
				every_county_province = { barony = { set_color_from_title = title:k_armenia } }
			}
			else_if = {
				limit = { holder = { has_trait = pro_persian } }
				every_county_province = { barony = { set_color_from_title = title:k_persia } }
			}
		}
		remove_global_variable = custom_map_mode_changing
	}
	is_shown = {
		exists = global_var:custom_map_mode
		global_var:custom_map_mode = flag:armenian_struggle_realm_alignment_map
	}
	is_valid = {
		NOT = { exists = global_var:custom_map_mode_changing }
	}
}

minority_any_map_mode = {
	effect = {
		root = { save_temporary_scope_as = the_culture }
		set_global_variable = { name = custom_map_mode_changing value = 1 }
		set_global_variable = { name = custom_map_mode value = flag:minority_cultures_map_mode }
		every_county = {
			if = {
				limit = {
					county_has_minority_faith_trigger = yes
					county_has_minority_culture_trigger = yes
				}
				every_county_province = { barony = { set_color_from_title = title:k_andalusia } }
			}
			else_if = {
				limit = {
					county_has_minority_faith_trigger = yes
					county_has_minority_culture_trigger = no
				}
				every_county_province = { barony = { set_color_from_title = title:k_france } }
			}
			else_if = {
				limit = {
					county_has_minority_faith_trigger = no
					county_has_minority_culture_trigger = yes
				}
				every_county_province = { barony = { set_color_from_title = title:k_castille } }
			}
			else = {
				every_county_province = { barony = { set_color_from_title = title:d_map_base_color } }
			}
		}
		remove_global_variable = custom_map_mode_changing
	}
	is_shown = {
		exists = global_var:custom_map_mode
		global_var:custom_map_mode = flag:minority_cultures_map_mode
	}
	is_valid = {
		NOT = { exists = global_var:custom_map_mode_changing }
	}
}

sea_languages_map_mode = {
    effect = {
		set_global_variable = { name = custom_map_mode_changing value = 1 }
		set_global_variable = { name = custom_map_mode value = flag:sea_languages_map_mode }

		every_county = {
			every_county_province = { barony = { set_color_from_title = title:d_map_base_color } }
		}
		every_county = {

			culture = {
				save_scope_as = this_culture

				if = {
					limit = {
						# is any culture sharing this language already in the list
						any_in_global_list = {
							variable = languages_map_cultures_list
							has_same_culture_language = scope:this_culture
						}
					}

					random_in_global_list = {
						variable = languages_map_cultures_list
						limit = {
							has_same_culture_language = scope:this_culture
						}

						var:language_color_title = { save_scope_as = language_color_title }
					}
				}
				else = {
					random_kingdom = {
						save_scope_as = language_color_title
					}

					set_variable = {
						name = language_color_title
						value = scope:language_color_title
					}

					add_to_global_variable_list = {
						name = languages_map_cultures_list
						target = scope:this_culture
					}
				}

				every_culture_county = {
					every_county_province = { barony = { set_color_from_title = scope:language_color_title } }
				}
			}
		}
        remove_global_variable = custom_map_mode_changing
    }
	is_shown = {
		exists = global_var:custom_map_mode
		global_var:custom_map_mode = flag:sea_languages_map_mode
	}
	is_valid = { NOT = { exists = global_var:custom_map_mode_changing } }
}
