
five_year_playable_pulse = {
	on_actions = {
		delay = { days = { 30 330 } }
		spawn_minority_characters
	}
}

# County changes faith.
# Root is the county title
# scope:old_faith is the faith it used to have
on_county_faith_change = {
	events = {
		false_conversion.0002	# Flag possible false conversions.
		false_conversion.0003	# Cleanup for other events.
		minorities.3001
	}
}

# Root = County title
# scope:old_culture = old culture
on_county_culture_change = {
	events = {
		minorities.3002
	}
}

do_culture_faith_migration = {
	effect = {
		if = {
			limit = {
				scope:source_county = {
					NOR = {
						culture = { has_cultural_tradition = tradition_isolationist }
						faith = { has_doctrine = tenet_communal_identity }
					}
				}
			}
			county_to_county_culture_faith_migration_effect = yes
		}
	}
}
do_minority_culture_growth = {
	effect = {
		scope:source_county = {

			every_in_list = {
				variable = culture_minorities_small
				add_to_temporary_list = potential_cultures
			}
			every_in_list = {
				variable = culture_minorities_large
				add_to_temporary_list = potential_cultures
			}
			every_in_list = {
				variable = culture_minorities_large
				add_to_temporary_list = potential_cultures
			}

			random_in_list = {
				list = potential_cultures
				save_scope_as = minority_culture
				random_list = {
					50 = { minority_culture_growth_effect = yes }
					50 = { minority_culture_spread_effect = yes }
				}
			}
		}
	}
}
do_minority_faith_growth = {

# Root is a count+ character
spawn_minority_characters = {
	trigger = {
		exists = root
		exists = root.capital_province
		any_sub_realm_county = {
			OR = {
				county_has_minority_culture_trigger = yes
				county_has_minority_faith_trigger = yes
			}
		}
	}
	effect = {
		every_courtier_or_guest = {
			culture = { add_to_temporary_list = represented_cultures }
			faith = { add_to_temporary_list = represented_faiths }
		}

		### Get a culture-faith combo based on a minority faith
		every_held_title = {
			limit = {
				tier = tier_county
				county_has_minority_faith_trigger = yes
				OR = {
					any_in_list = {
						variable = faith_minorities_small
						NOT = { is_in_list = represented_faiths }
						# Temporary, don't currently want to ask if Jews can/should be other cultures/faiths and they're generally covered by base game mechanics anyway
						is_endogamous_faith_trigger = no
					}
					any_in_list = {
						variable = faith_minorities_large
						NOT = { is_in_list = represented_faiths }
						is_endogamous_faith_trigger = no
					}
				}
			}

			## First get an unrepresented minority faith
			every_in_list = {
				variable = faith_minorities_large
				if = {
					limit = {
						NOR = { is_in_list = represented_faiths }
						is_endogamous_faith_trigger = no
					}
					add_to_temporary_list = potential_faiths
					add_to_temporary_list = potential_faiths
				}
			}
			every_in_list = {
				variable = faith_minorities_small
				if = {
					limit = {
						NOR = { is_in_list = represented_faiths }
						is_endogamous_faith_trigger = no
					}
					add_to_temporary_list = potential_faiths
				}
			}

			## Next get any present culture
			culture = {
				add_to_temporary_list = potential_cultures
				add_to_temporary_list = potential_cultures
				add_to_temporary_list = potential_cultures
				add_to_temporary_list = potential_cultures
			}
			if = {
				limit = { county_has_non_endogamous_large_minority_culture_trigger = yes }
				every_in_list = {
					variable = culture_minorities_large
					add_to_temporary_list = potential_cultures
					add_to_temporary_list = potential_cultures
				}
			}
			if = {
				limit = { county_has_non_endogamous_small_minority_culture_trigger = yes }
				every_in_list = {
					variable = culture_minorities_small
					add_to_temporary_list = potential_cultures
				}
			}
		}

		if = {
			limit = {
				list_size = { list = potential_cultures value > 0 }
				list_size = { list = potential_faiths value > 0 }
			}
			random_in_list = {
				list = potential_cultures
				save_scope_as = culture_to_spawn_option_1
			}
			random_in_list = {
				list = potential_faiths
				save_scope_as = faith_to_spawn_option_1
			}
		}

		clear_variable_list = potential_cultures
		clear_variable_list = potential_faiths

		### Next get a culture-faith combo based on a minority culture
		every_held_title = {
			limit = {
				tier = tier_county
				county_has_minority_culture_trigger = yes
				OR = {
					any_in_list = {
						variable = culture_minorities_small
						NOT = { is_in_list = represented_cultures }
						# Temporary, don't currently want to ask if Jews can/should be other faiths/cultures and they're generally covered by base game mechanics anyway
						is_endogamous_culture_trigger = no
					}
					any_in_list = {
						variable = culture_minorities_large
						NOT = { is_in_list = represented_cultures }
						is_endogamous_culture_trigger = no
					}
				}
			}

			## First get an unrepresented minority culture
			every_in_list = {
				variable = culture_minorities_large
				if = {
					limit = {
						NOR = { is_in_list = represented_cultures }
						is_endogamous_culture_trigger = no
					}
					add_to_temporary_list = potential_cultures
					add_to_temporary_list = potential_cultures
				}
			}
			every_in_list = {
				variable = culture_minorities_small
				if = {
					limit = {
						NOR = { is_in_list = represented_cultures }
						is_endogamous_culture_trigger = no
					}
					add_to_temporary_list = potential_cultures
				}
			}

			## Next get any present faith
			faith = {
				add_to_temporary_list = potential_faiths
				add_to_temporary_list = potential_faiths
				add_to_temporary_list = potential_faiths
				add_to_temporary_list = potential_faiths
			}
			if = {
				limit = { county_has_non_endogamous_large_minority_faith_trigger = yes }
				every_in_list = {
					variable = faith_minorities_large
					add_to_temporary_list = potential_faiths
					add_to_temporary_list = potential_faiths
				}
			}
			if = {
				limit = { county_has_non_endogamous_small_minority_faith_trigger = yes }
				every_in_list = {
					variable = faith_minorities_small
					add_to_temporary_list = potential_faiths
				}
			}
		}

		if = {
			limit = {
				list_size = { list = potential_faiths value > 0 }
				list_size = { list = potential_cultures value > 0 }
			}
			random_in_list = {
				list = potential_faiths
				save_scope_as = faith_to_spawn_option_2
			}
			random_in_list = {
				list = potential_cultures
				save_scope_as = culture_to_spawn_option_2
			}
		}

		clear_variable_list = potential_faiths
		clear_variable_list = potential_cultures

		random_in_list = {
			list = potential_cultures
			save_scope_as = culture_to_spawn_option_2
		}
		random_in_list = {
			list = potential_faiths
			save_scope_as = faith_to_spawn_option_2
		}

		random_list = {
			50 = {
				trigger = {
					exists = scope:culture_to_spawn_option_1
					exists = scope:faith_to_spawn_option_1
				}

				create_character = {
					location = root.location
					age = { 18 45 }
					gender_female_chance = 50
					random_traits = yes
					faith = scope:faith_to_spawn_option_1
					culture = scope:culture_to_spawn_option_1
					save_scope_as = minority_courtier
				}

				add_visiting_courtier = scope:minority_courtier
				send_interface_toast = {
					type = character_arrived
					left_icon = scope:minority_courtier
					title = local_minority_courtier_arrived.t
					desc = local_minority_courtier_arrived.desc
				}
			}
			50 = {
				trigger = {
					exists = scope:culture_to_spawn_option_2
					exists = scope:faith_to_spawn_option_2
				}

				create_character = {
					location = root.location
					age = { 18 45 }
					gender_female_chance = 50
					random_traits = yes
					faith = scope:faith_to_spawn_option_2
					culture = scope:culture_to_spawn_option_2
					save_scope_as = minority_courtier
				}

				add_visiting_courtier = scope:minority_courtier
				send_interface_toast = {
					type = character_arrived
					left_icon = scope:minority_courtier
					title = local_minority_courtier_arrived.t
					desc = local_minority_courtier_arrived.desc
				}
			}
		}

		clear_variable_list = potential_cultures
		clear_variable_list = potential_faiths
		clear_saved_scope = culture_to_spawn_option_1
		clear_saved_scope = faith_to_spawn_option_1
		clear_saved_scope = culture_to_spawn_option_2
		clear_saved_scope = faith_to_spawn_option_2
	}
}
